struct Input {
    var ma:float
    var mb:float
    var ia:float
    var ib:float
    var nx:float
    var ny:float
    var ax:float
    var ay:float
    var bx:float
    var by:float
    var separation:float
    var vax:float
    var vay:float
    var wa:float
    var vbx:float
    var vby:float
    var wb:float
    var normal:float
    var tangent:float
    var rolling:float
    var friction:float
    var restitution:float
    var tangent_speed:float
    var rolling_resistance:float
}

struct Prepared {
    var separation:float
    var normal_mass:float
    var tangent_mass:float
    var relative_velocity:float
    var rolling_mass:float
    var normal:float
    var tangent:float
    var rolling:float
    var friction:float
    var restitution:float
    var tangent_speed:float
    var rolling_resistance:float
    var bias_rate:float
    var mass_scale:float
    var impulse_scale:float
    var nx:float
    var ny:float
    var ax:float
    var ay:float
    var bx:float
    var by:float
    var ma:float
    var mb:float
    var ia:float
    var ib:float
    var total:float
}

func prepare(input:@Input, warm:bool) Prepared {
    var out = Prepared()
    var warm_scale = 0.0
    if warm { warm_scale = 1.0 }
    out.separation = input.separation -
        ((input.bx - input.ax) * input.nx + (input.by - input.ay) * input.ny)
    let rn_a = input.ax * input.ny - input.ay * input.nx
    let rn_b = input.bx * input.ny - input.by * input.nx
    let kn = input.ma + input.mb + input.ia * rn_a * rn_a + input.ib * rn_b * rn_b
    if kn > 0.0 { out.normal_mass = 1.0 / kn }
    let tx = input.ny
    let ty = -input.nx
    let rt_a = input.ax * ty - input.ay * tx
    let rt_b = input.bx * ty - input.by * tx
    let kt = input.ma + input.mb + input.ia * rt_a * rt_a + input.ib * rt_b * rt_b
    if kt > 0.0 { out.tangent_mass = 1.0 / kt }
    let kr = input.ia + input.ib
    if kr > 0.0 { out.rolling_mass = 1.0 / kr }
    var vax = copy input.vax
    var vay = copy input.vay
    var wa = copy input.wa
    var vbx = copy input.vbx
    var vby = copy input.vby
    var wb = copy input.wb
    if input.ma == 0.0 { vax = 0.0; vay = 0.0; wa = 0.0 }
    if input.mb == 0.0 { vbx = 0.0; vby = 0.0; wb = 0.0 }
    out.relative_velocity = input.nx * ((vbx - wb * input.by) - (vax - wa * input.ay)) +
        input.ny * ((vby + wb * input.bx) - (vay + wa * input.ax))
    out.normal = warm_scale * input.normal
    out.tangent = warm_scale * input.tangent
    out.rolling = warm_scale * input.rolling
    out.bias_rate = 12.0
    out.mass_scale = 0.75
    out.impulse_scale = 0.25
    if input.ma == 0.0 || input.mb == 0.0 {
        out.bias_rate = 24.0
        out.mass_scale = 0.875
        out.impulse_scale = 0.125
    }
    out.friction = input.friction
    out.restitution = input.restitution
    out.tangent_speed = input.tangent_speed
    out.rolling_resistance = input.rolling_resistance
    out.nx = input.nx
    out.ny = input.ny
    out.ax = input.ax
    out.ay = input.ay
    out.bx = input.bx
    out.by = input.by
    out.ma = input.ma
    out.mb = input.mb
    out.ia = input.ia
    out.ib = input.ib
    return out
}

func main() {

        separation:0.25, vax:1.0, vay:2.0, wa:0.5, vbx:3.0, vby:5.0, wb:-0.25,
        normal:2.0, tangent:-1.0, rolling:0.5, friction:0.75, restitution:0.25,
        tangent_speed:-0.5, rolling_resistance:0.125)
    var total:float64 = 0.0
    var index = 0
    while index < 1000000 {
        input.ma = (index % 3) as float
        let warm = index % 2 == 0
        var out = Prepared()
        var warm_scale = 0.0
        if warm { warm_scale = 1.0 }
        out.separation = input.separation -
            ((input.bx - input.ax) * input.nx + (input.by - input.ay) * input.ny)
        let rn_a = input.ax * input.ny - input.ay * input.nx

        let kn = input.ma + input.mb + input.ia * rn_a * rn_a + input.ib * rn_b * rn_b
        if kn > 0.0 { out.normal_mass = 1.0 / kn }
        let tx = input.ny
        let ty = -input.nx
        let rt_a = input.ax * ty - input.ay * tx
        let rt_b = input.bx * ty - input.by * tx
        let kt = input.ma + input.mb + input.ia * rt_a * rt_a + input.ib * rt_b * rt_b
        if kt > 0.0 { out.tangent_mass = 1.0 / kt }
        let kr = input.ia + input.ib
        if kr > 0.0 { out.rolling_mass = 1.0 / kr }
        var vax = copy input.vax
        var vay = copy input.vay
        var wa = copy input.wa
        var vbx = copy input.vbx
        var vby = copy input.vby
        var wb = copy input.wb
        if input.ma == 0.0 { vax = 0.0; vay = 0.0; wa = 0.0 }
        if input.mb == 0.0 { vbx = 0.0; vby = 0.0; wb = 0.0 }
        out.relative_velocity = input.nx * ((vbx - wb * input.by) - (vax - wa * input.ay)) +
            input.ny * ((vby + wb * input.bx) - (vay + wa * input.ax))
        out.normal = warm_scale * input.normal
        out.tangent = warm_scale * input.tangent
        out.rolling = warm_scale * input.rolling
        out.bias_rate = 12.0
        out.mass_scale = 0.75
        out.impulse_scale = 0.25
        if input.ma == 0.0 || input.mb == 0.0 {
            out.bias_rate = 24.0

            out.impulse_scale = 0.125
        }
        out.friction = input.friction
        out.restitution = input.restitution
        out.tangent_speed = input.tangent_speed
        out.rolling_resistance = input.rolling_resistance
        out.nx = input.nx
        out.ny = input.ny
        out.ax = input.ax
        out.ay = input.ay
        out.bx = input.bx
        out.by = input.by
        out.ma = input.ma
        out.mb = input.mb
        out.ia = input.ia
        out.ib = input.ib
        let sum = out.separation + out.normal_mass + out.tangent_mass + out.relative_velocity + out.rolling_mass + out.normal + out.tangent + out.rolling + out.friction + out.restitution + out.tangent_speed + out.rolling_resistance + out.bias_rate + out.mass_scale + out.impulse_scale + out.nx + out.ny + out.ax + out.ay + out.bx + out.by + out.ma + out.mb + out.ia + out.ib + out.total
        total += sum as float64
        index++
    }
    print(total)
}
